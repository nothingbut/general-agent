// v3/src/GeneralAgent.Core/Models/SessionTag.cs
namespace GeneralAgent.Core.Models;

/// <summary>
/// 会话标签实体（不可变）
/// </summary>
public sealed record SessionTag
{
    /// <summary>
    /// 会话 ID
    /// </summary>
    public Guid SessionId { get; init; }

    /// <summary>
    /// 标签名称（小写存储）
    /// </summary>
    public string Tag { get; init; } = string.Empty;

    /// <summary>
    /// 标签颜色（RGB 格式，如 "#FF5733"）
    /// </summary>
    public string? Color { get; init; }

    /// <summary>
    /// 标签图标（Emoji）
    /// </summary>
    public string? Emoji { get; init; }

    /// <summary>
    /// 创建时间（UTC）
    /// </summary>
    public DateTime CreatedAt { get; init; }

    /// <summary>
    /// 标签来源
    /// </summary>
    public TagSource Source { get; init; } = TagSource.User;

    /// <summary>
    /// 创建新标签
    /// </summary>
    public static SessionTag Create(
        Guid sessionId,
        string tag,
        TagSource source = TagSource.User,
        string? color = null,
        string? emoji = null)
        => new()
        {
            SessionId = sessionId,
            Tag = tag.Trim().ToLowerInvariant(),
            Color = color,
            Emoji = emoji,
            CreatedAt = DateTime.UtcNow,
            Source = source
        };
}

/// <summary>
/// 标签来源
/// </summary>
public enum TagSource
{
    User,  // 用户手动添加
    Auto   // 系统自动生成
}

namespace GeneralAgent.Core.Models;

/// <summary>
/// 搜索查询模型
/// </summary>
public sealed record SearchQuery
{
    /// <summary>
    /// 原始自然语言查询
    /// </summary>
    public string NaturalQuery { get; init; } = string.Empty;

    /// <summary>
    /// 结构化查询条件
    /// </summary>
    public SearchCriteria Criteria { get; init; } = new();

    /// <summary>
    /// 查询类型
    /// </summary>
    public SearchType Type { get; init; } = SearchType.Keyword;
}

/// <summary>
/// 搜索条件
/// </summary>
public sealed record SearchCriteria
{
    /// <summary>
    /// 关键词（用于 LIKE 查询）
    /// </summary>
    public List<string> Keywords { get; init; } = new();

    /// <summary>
    /// 正则表达式模式
    /// </summary>
    public string? RegexPattern { get; init; }

    /// <summary>
    /// 精确短语
    /// </summary>
    public List<string> ExactPhrases { get; init; } = new();

    /// <summary>
    /// 角色过滤
    /// </summary>
    public MessageRole? Role { get; init; }

    /// <summary>
    /// 会话 ID 过滤
    /// </summary>
    public Guid? SessionId { get; init; }

    /// <summary>
    /// 时间范围过滤
    /// </summary>
    public DateTimeOffset? StartDate { get; init; }
    public DateTimeOffset? EndDate { get; init; }
}

/// <summary>
/// 搜索类型
/// </summary>
public enum SearchType
{
    Keyword,      // 关键词搜索
    Regex,        // 正则表达式
    Exact,        // 精确匹配
    Semantic      // 语义搜索（未来扩展）
}

// v3/src/GeneralAgent.Infrastructure/Storage/Repositories/SessionTagRepository.cs
namespace GeneralAgent.Infrastructure.Storage.Repositories;

/// <summary>
/// SessionTag 仓储实现
/// </summary>
public sealed class SessionTagRepository : ISessionTagRepository
{
    private readonly AgentDbContext _context;

    public SessionTagRepository(AgentDbContext context)
    {
        _context = context ?? throw new ArgumentNullException(nameof(context));
    }

    public async Task AddAsync(SessionTag tag, CancellationToken ct = default)
    {
        try
        {
            _context.Set<SessionTag>().Add(tag);
            await _context.SaveChangesAsync(ct);
        }
        catch (Exception ex)
        {
            throw new StorageException($"Failed to add tag: {ex.Message}", ex);
        }
    }

    public async Task RemoveAsync(Guid sessionId, string tag, CancellationToken ct = default)
    {
        try
        {
            var entity = await _context.Set<SessionTag>()
                .FirstOrDefaultAsync(
                    t => t.SessionId == sessionId && t.Tag == tag.ToLowerInvariant(),
                    ct
                );

            if (entity != null)
            {
                _context.Set<SessionTag>().Remove(entity);
                await _context.SaveChangesAsync(ct);
            }
        }
        catch (Exception ex)
        {
            throw new StorageException($"Failed to remove tag: {ex.Message}", ex);
        }
    }

    public async Task<List<SessionTag>> GetBySessionAsync(
        Guid sessionId,
        CancellationToken ct = default)
    {
        try
        {
            return await _context.Set<SessionTag>()
                .AsNoTracking()
                .Where(t => t.SessionId == sessionId)
                .OrderBy(t => t.CreatedAt)
                .ToListAsync(ct);
        }
        catch (Exception ex)
        {
            throw new StorageException($"Failed to get tags by session: {ex.Message}", ex);
        }
    }

    public async Task<List<Guid>> GetSessionsByTagAsync(
        string tag,
        CancellationToken ct = default)
    {
        try
        {
            return await _context.Set<SessionTag>()
                .AsNoTracking()
                .Where(t => t.Tag == tag.ToLowerInvariant())
                .Select(t => t.SessionId)
                .Distinct()
                .ToListAsync(ct);
        }
        catch (Exception ex)
        {
            throw new StorageException($"Failed to get sessions by tag: {ex.Message}", ex);
        }
    }

    public async Task<Dictionary<string, int>> GetTagStatisticsAsync(
        CancellationToken ct = default)
    {
        try
        {
            return await _context.Set<SessionTag>()
                .AsNoTracking()
                .GroupBy(t => t.Tag)
                .Select(g => new { Tag = g.Key, Count = g.Count() })
                .ToDictionaryAsync(x => x.Tag, x => x.Count, ct);
        }
        catch (Exception ex)
        {
            throw new StorageException($"Failed to get tag statistics: {ex.Message}", ex);
        }
    }
}

    {
        builder.ToTable("session_tags");

        // 复合主键
        builder.HasKey(t => new { t.SessionId, t.Tag });

        builder.Property(t => t.Tag)
            .HasMaxLength(50)
            .IsRequired();

        builder.Property(t => t.Color)
            .HasMaxLength(7);

        builder.Property(t => t.Emoji)
            .HasMaxLength(10);

        builder.Property(t => t.CreatedAt)
            .IsRequired();

        builder.Property(t => t.Source)
            .HasConversion<string>()
            .HasMaxLength(20)
            .IsRequired();

        // 索引
        builder.HasIndex(t => t.Tag);
        builder.HasIndex(t => t.Source);

        // 外键（级联删除）
        builder.HasOne<Session>()
            .WithMany()
            .HasForeignKey(t => t.SessionId)
            .OnDelete(DeleteBehavior.Cascade);

            CREATE TRIGGER messages_ai AFTER INSERT ON messages BEGIN

                VALUES (new.id, new.session_id, new.content, new.role, new.created_at);
            END;
        ");

            CREATE TRIGGER messages_ad AFTER DELETE ON messages BEGIN
                DELETE FROM messages_fts WHERE message_id = old.id;
            END;
        ");

        migrationBuilder.Sql(@"
            CREATE TRIGGER messages_au AFTER UPDATE ON messages BEGIN
                DELETE FROM messages_fts WHERE message_id = old.id;
                INSERT INTO messages_fts(message_id, session_id, content, role, created_at)
                VALUES (new.id, new.session_id, new.content, new.role, new.created_at);
            END;

            SELECT id, session_id, content, role, created_at FROM messages;

    {
        migrationBuilder.DropTable("session_tags");

        migrationBuilder.Sql("DROP TRIGGER IF EXISTS messages_ai;");
        migrationBuilder.Sql("DROP TRIGGER IF EXISTS messages_ad;");

// v3/src/GeneralAgent.Infrastructure/Caching/SearchQueryCache.cs
namespace GeneralAgent.Infrastructure.Caching;

/// <summary>
/// 搜索查询 LRU 缓存实现
/// </summary>
public sealed class SearchQueryCache : ISearchQueryCache
{
    private readonly LinkedList<CacheEntry> _lruList = new();
    private readonly Dictionary<string, LinkedListNode<CacheEntry>> _cache = new();
    private readonly int _capacity;
    private readonly TimeSpan _ttl;
    private readonly object _lock = new();

    public SearchQueryCache(int capacity = 100, TimeSpan? ttl = null)
    {
        _capacity = capacity;
        _ttl = ttl ?? TimeSpan.FromHours(1);
    }

    public SearchQuery? Get(string naturalQuery)
    {
        lock (_lock)
        {
            if (_cache.TryGetValue(naturalQuery, out var node))
            {
                // 检查是否过期
                if (DateTime.UtcNow - node.Value.Timestamp < _ttl)
                {
                    // 移到链表头部（标记为最近使用）
                    _lruList.Remove(node);
                    _lruList.AddFirst(node);
                    return node.Value.Query;
                }

                // 过期，移除
                _cache.Remove(naturalQuery);
                _lruList.Remove(node);
            }

            return null;
        }
    }

    public void Set(string naturalQuery, SearchQuery searchQuery)
    {
        lock (_lock)
        {
            if (_cache.TryGetValue(naturalQuery, out var existingNode))
            {
                // 更新现有项
                existingNode.Value = new CacheEntry(searchQuery, DateTime.UtcNow);
                _lruList.Remove(existingNode);
                _lruList.AddFirst(existingNode);
            }
            else
            {
                // 添加新项
                if (_cache.Count >= _capacity)
                {
                    // 淘汰最久未使用的项
                    var lastNode = _lruList.Last!;
                    _cache.Remove(lastNode.Value.Query.NaturalQuery);
                    _lruList.RemoveLast();
                }

                var newNode = new LinkedListNode<CacheEntry>(
                    new CacheEntry(searchQuery, DateTime.UtcNow)
                );
                _lruList.AddFirst(newNode);
                _cache[naturalQuery] = newNode;
            }
        }
    }

    public void Clear()
    {
        lock (_lock)
        {
            _cache.Clear();
            _lruList.Clear();
        }
    }

    private record CacheEntry(SearchQuery Query, DateTime Timestamp);
}
```

// v3/src/GeneralAgent.Application/Services/NaturalLanguageQueryService.cs
namespace GeneralAgent.Application.Services;

/// <summary>
/// 自然语言查询服务
/// </summary>
public sealed class NaturalLanguageQueryService
{
    private readonly ILLMClient _llmClient;
    private readonly ISearchQueryCache _queryCache;
    private readonly ILogger<NaturalLanguageQueryService> _logger;

    public NaturalLanguageQueryService(
        ILLMClient llmClient,
        ISearchQueryCache queryCache,
        ILogger<NaturalLanguageQueryService> logger)
    {
        _llmClient = llmClient;
        _queryCache = queryCache;
        _logger = logger;
    }

    /// <summary>
    /// 解析自然语言查询
    /// </summary>
    public async Task<SearchQuery> ParseQueryAsync(
        string naturalQuery,
        CancellationToken ct = default)
    {
        // 1. 检查缓存
        var cachedQuery = _queryCache.Get(naturalQuery);
        if (cachedQuery != null)
        {
            _logger.LogDebug("查询缓存命中: {Query}", naturalQuery);
            return cachedQuery;
        }

        // 2. 调用 LLM 解析
        var prompt = BuildParsePrompt(naturalQuery);

        using var timeoutCts = new CancellationTokenSource(TimeSpan.FromSeconds(5));
        using var linkedCts = CancellationTokenSource.CreateLinkedTokenSource(ct, timeoutCts.Token);

        try
        {
            var response = await _llmClient.GenerateAsync(prompt, linkedCts.Token);
            var criteria = ParseLLMResponse(response);

            var query = new SearchQuery
            {
                NaturalQuery = naturalQuery,
                Criteria = criteria,
                Type = DetermineSearchType(criteria)
            };

            // 3. 缓存结果
            _queryCache.Set(naturalQuery, query);
            return query;
        }
        catch (OperationCanceledException) when (timeoutCts.IsCancellationRequested)
        {
            _logger.LogWarning("查询解析超时，回退到关键词搜索");
            return FallbackToKeywordSearch(naturalQuery);
        }
        catch (JsonException ex)
        {
            _logger.LogWarning(ex, "LLM 返回无效 JSON，回退到关键词搜索");
            return FallbackToKeywordSearch(naturalQuery);
        }
    }

    private string BuildParsePrompt(string naturalQuery)
    {
        return $@"
你是一个搜索查询解析器。将用户的自然语言查询转换为结构化的搜索条件。

用户查询: {naturalQuery}

请分析查询并返回 JSON 格式的搜索条件：
{{
  ""keywords"": [""关键词1"", ""关键词2""],
  ""exactPhrases"": [""精确短语""],
  ""regexPattern"": ""正则表达式（如果需要）"",
  ""role"": ""user"" | ""assistant"" | null,
  ""sessionId"": ""会话ID"" | null,
  ""startDate"": ""ISO8601日期"" | null,
  ""endDate"": ""ISO8601日期"" | null
}}

示例：
- ""查找昨天关于 Python 的讨论"" → {{""keywords"":[""Python""], ""startDate"":""2026-03-24T00:00:00Z""}}
- ""搜索我问的所有问题"" → {{""keywords"":[], ""role"":""user""}}
- ""包含 'bug fix' 的消息"" → {{""exactPhrases"":[""bug fix""]}}

只返回 JSON，不要其他解释。
";
    }

    private SearchCriteria ParseLLMResponse(string response)
    {
        // 清理可能的 Markdown 代码块
        var json = response.Trim();
        if (json.StartsWith("```json"))
        {
            json = json.Substring(7);
        }
        if (json.StartsWith("```"))
        {
            json = json.Substring(3);
        }
        if (json.EndsWith("```"))
        {
            json = json.Substring(0, json.Length - 3);
        }
        json = json.Trim();

        var doc = JsonDocument.Parse(json);
        var root = doc.RootElement;

        var criteria = new SearchCriteria();

        if (root.TryGetProperty("keywords", out var keywords))
        {
            criteria = criteria with
            {
                Keywords = keywords.EnumerateArray()
                    .Select(k => k.GetString()!)
                    .Where(k => !string.IsNullOrEmpty(k))
                    .ToList()
            };
        }

        if (root.TryGetProperty("exactPhrases", out var phrases))
        {
            criteria = criteria with
            {
                ExactPhrases = phrases.EnumerateArray()
                    .Select(p => p.GetString()!)
                    .Where(p => !string.IsNullOrEmpty(p))
                    .ToList()
            };
        }

        if (root.TryGetProperty("regexPattern", out var regex) &&
            regex.ValueKind != JsonValueKind.Null)
        {
            criteria = criteria with { RegexPattern = regex.GetString() };
        }

        if (root.TryGetProperty("role", out var role) &&
            role.ValueKind != JsonValueKind.Null)
        {
            var roleStr = role.GetString();
            if (Enum.TryParse<MessageRole>(roleStr, true, out var roleEnum))
            {
                criteria = criteria with { Role = roleEnum };
            }
        }

        if (root.TryGetProperty("startDate", out var startDate) &&
            startDate.ValueKind != JsonValueKind.Null)
        {
            if (DateTimeOffset.TryParse(startDate.GetString(), out var start))
            {
                criteria = criteria with { StartDate = start };
            }
        }

        if (root.TryGetProperty("endDate", out var endDate) &&
            endDate.ValueKind != JsonValueKind.Null)
        {
            if (DateTimeOffset.TryParse(endDate.GetString(), out var end))
            {
                criteria = criteria with { EndDate = end };
            }
        }

        return criteria;
    }

    private SearchType DetermineSearchType(SearchCriteria criteria)
    {
        if (!string.IsNullOrEmpty(criteria.RegexPattern))
            return SearchType.Regex;

        if (criteria.ExactPhrases.Any())
            return SearchType.Exact;

        return SearchType.Keyword;
    }

    private SearchQuery FallbackToKeywordSearch(string query)
    {
        return new SearchQuery
        {
            NaturalQuery = query,
            Criteria = new SearchCriteria
            {
                Keywords = new List<string> { query }
            },
            Type = SearchType.Keyword
        };
    }
}

/// <summary>
/// 智能标签建议服务（实现 ISmartTagService 接口）
/// </summary>
public sealed class SmartTagService : ISmartTagService
{
    private const int MaxTagsPerSession = 5;
    private const int MaxSuggestionsFromTitle = 3;
    private const int MaxSuggestionsFromContent = 5;

    private readonly ILLMClient _llmClient;
    private readonly ISessionTagRepository _tagRepository;
    private readonly ILogger<SmartTagService> _logger;

    public SmartTagService(
        ILLMClient llmClient,
        ISessionTagRepository tagRepository,
        ILogger<SmartTagService> logger)
    {
        _llmClient = llmClient;
        _tagRepository = tagRepository;
        _logger = logger;
    }

    /// <summary>
    /// 基于标题生成标签建议（快速模式）
    /// </summary>
    public async Task<List<TagSuggestion>> SuggestFromTitleAsync(
        string title,
        CancellationToken ct = default)
    {
        var prompt = BuildTitlePrompt(title, MaxSuggestionsFromTitle);

        using var timeoutCts = new CancellationTokenSource(TimeSpan.FromSeconds(3));
        using var linkedCts = CancellationTokenSource.CreateLinkedTokenSource(ct, timeoutCts.Token);

        try
        {
            var response = await _llmClient.GenerateAsync(prompt, linkedCts.Token);
            return ParseTagSuggestions(response);
        }
        catch (OperationCanceledException) when (timeoutCts.IsCancellationRequested)
        {
            _logger.LogWarning("标签生成超时（标题模式）");
            return new List<TagSuggestion>();
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "标签生成失败（标题模式）");
            return new List<TagSuggestion>();
        }
    }

    /// <summary>
    /// 基于内容生成标签建议（深度分析模式）
    /// </summary>
    public async Task<List<TagSuggestion>> SuggestFromContentAsync(
        Guid sessionId,
        List<Message> messages,
        CancellationToken ct = default)
    {
        var prompt = BuildContentPrompt(messages, MaxSuggestionsFromContent);

        using var timeoutCts = new CancellationTokenSource(TimeSpan.FromSeconds(10));
        using var linkedCts = CancellationTokenSource.CreateLinkedTokenSource(ct, timeoutCts.Token);

        try
        {
            var response = await _llmClient.GenerateAsync(prompt, linkedCts.Token);
            return ParseTagSuggestions(response);
        }
        catch (OperationCanceledException) when (timeoutCts.IsCancellationRequested)
        {
            _logger.LogWarning("标签生成超时（内容模式）");
            return new List<TagSuggestion>();
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "标签生成失败（内容模式）");
            return new List<TagSuggestion>();
        }
    }

    /// <summary>
    /// 应用标签建议（去重、限额检查）
    /// </summary>
    public async Task ApplySuggestionsAsync(
        Guid sessionId,
        List<TagSuggestion> suggestions,
        CancellationToken ct = default)
    {
        // 1. 获取现有标签
        var existingTags = await _tagRepository.GetBySessionAsync(sessionId, ct);
        var existingTagNames = existingTags.Select(t => t.Tag.ToLowerInvariant()).ToHashSet();

        // 2. 过滤重复和超限
        var availableSlots = MaxTagsPerSession - existingTags.Count;
        if (availableSlots <= 0)
        {
            _logger.LogWarning("会话 {SessionId} 标签数量已达上限 {Max}", sessionId, MaxTagsPerSession);
            return;
        }

        var newSuggestions = suggestions
            .Where(s => !existingTagNames.Contains(s.Tag.ToLowerInvariant()))
            .Take(availableSlots)
            .ToList();

        // 3. 添加标签
        foreach (var suggestion in newSuggestions)
        {
            var tag = SessionTag.Create(
                sessionId,
                suggestion.Tag,
                TagSource.Auto,
                suggestion.Color,
                suggestion.Emoji
            );
            await _tagRepository.AddAsync(tag, ct);
        }

        _logger.LogInformation(
            "为会话 {SessionId} 添加了 {Count} 个自动标签",
            sessionId,
            newSuggestions.Count
        );
    }

会话标题: {title}

颜色格式：RGB 十六进制（如 #FF5733）
Emoji：选择最相关的表情符号

只返回 JSON，不要其他解释。

    }

    private string BuildContentPrompt(List<Message> messages, int maxCount)
    {
        var contentSample = string.Join("\n", messages.Take(10).Select(m => $"- {m.Content.Substring(0, Math.Min(100, m.Content.Length))}"));

        return $@"
你是一个标签生成专家。根据会话内容生成相关标签。

会话内容摘要（前10条消息）:
{contentSample}

请分析会话主题，生成 {maxCount} 个最相关的标签，包含名称、Emoji 和颜色。

返回 JSON 格式：
{{
  ""tags"": [
    {{""name"": ""标签名"", ""emoji"": ""📝"", ""color"": ""#FF5733""}},
    ...
  ]
}}

标签分类建议：
- 技术栈（python, react, docker 等）
- 主题（bug, feature, performance 等）
- 领域（ai, web, mobile 等）

只返回 JSON，不要其他解释。
";
    }

    private List<TagSuggestion> ParseTagSuggestions(string response)
    {
        try
        {
            // 清理 Markdown 代码块
            var json = response.Trim();
            if (json.StartsWith("```json")) json = json.Substring(7);
            if (json.StartsWith("```")) json = json.Substring(3);
            if (json.EndsWith("```")) json = json.Substring(0, json.Length - 3);
            json = json.Trim();

            var doc = JsonDocument.Parse(json);
            var tagsArray = doc.RootElement.GetProperty("tags");

            var suggestions = new List<TagSuggestion>();
            foreach (var tag in tagsArray.EnumerateArray())
            {
                var name = tag.GetProperty("name").GetString() ?? "";
                var emoji = tag.TryGetProperty("emoji", out var e) ? e.GetString() : null;
                var color = tag.TryGetProperty("color", out var c) ? c.GetString() : null;

                if (!string.IsNullOrEmpty(name))
                {
                    suggestions.Add(new TagSuggestion(name, emoji, color));
                }
            }

            return suggestions;
        }
        catch (JsonException ex)
        {
            _logger.LogWarning(ex, "解析标签建议失败");
            return new List<TagSuggestion>();
        }
    }

// v3/src/GeneralAgent.Application/Services/BackgroundTaskService.cs
namespace GeneralAgent.Application.Services;

/// <summary>
/// 后台任务服务（处理标签生成等异步任务）
/// </summary>
public sealed class BackgroundTaskService : BackgroundService
{
    private readonly Channel<TagSuggestionTask> _taskQueue;
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly ILogger<BackgroundTaskService> _logger;

    public BackgroundTaskService(
        IServiceScopeFactory scopeFactory,
        ILogger<BackgroundTaskService> logger)
    {
        _taskQueue = Channel.CreateUnbounded<TagSuggestionTask>();
        _scopeFactory = scopeFactory;
        _logger = logger;
    }

    /// <summary>
    /// 入队标签建议任务
    /// </summary>
    public async Task EnqueueTagSuggestionAsync(Guid sessionId)
    {
        var task = new TagSuggestionTask(sessionId, DateTime.UtcNow);
        await _taskQueue.Writer.WriteAsync(task);
        _logger.LogDebug("标签建议任务已入队: {SessionId}", sessionId);
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        _logger.LogInformation("后台任务服务已启动");

        await foreach (var task in _taskQueue.Reader.ReadAllAsync(stoppingToken))
        {
            try
            {
                await ProcessTagSuggestionAsync(task, stoppingToken);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "处理标签建议任务失败: {SessionId}", task.SessionId);
            }
        }

        _logger.LogInformation("后台任务服务已停止");
    }

    private async Task ProcessTagSuggestionAsync(TagSuggestionTask task, CancellationToken ct)
    {
        using var scope = _scopeFactory.CreateScope();
        var tagService = scope.ServiceProvider.GetRequiredService<SmartTagService>();
        var sessionRepo = scope.ServiceProvider.GetRequiredService<ISessionRepository>();
        var messageRepo = scope.ServiceProvider.GetRequiredService<IMessageRepository>();

        _logger.LogInformation("开始处理标签建议: {SessionId}", task.SessionId);

        // 1. 获取会话和消息
        var session = await sessionRepo.GetByIdAsync(task.SessionId, ct);
        if (session == null)
        {
            _logger.LogWarning("会话不存在: {SessionId}", task.SessionId);
            return;
        }

        var messages = await messageRepo.GetBySessionAsync(task.SessionId, ct);
        if (messages.Count == 0)
        {
            _logger.LogDebug("会话无消息，跳过标签建议: {SessionId}", task.SessionId);
            return;
        }

        // 2. 生成标签建议
        var suggestions = await tagService.SuggestFromContentAsync(task.SessionId, messages, ct);
        if (suggestions.Count == 0)
        {
            _logger.LogDebug("未生成标签建议: {SessionId}", task.SessionId);
            return;
        }

        // 3. 应用建议
        await tagService.ApplySuggestionsAsync(task.SessionId, suggestions, ct);

        _logger.LogInformation(
            "标签建议处理完成: {SessionId}, 建议数: {Count}",
            task.SessionId,
            suggestions.Count
        );
    }

    public override void Dispose()
    {
        _taskQueue.Writer.Complete();
        base.Dispose();
    }
}

/// <summary>
/// 标签建议任务
/// </summary>
internal sealed record TagSuggestionTask(Guid SessionId, DateTime EnqueuedAt);

// v3/src/GeneralAgent.Hosts.Console/Commands/TagCommand.cs
namespace GeneralAgent.Hosts.Console.Commands;

/// <summary>
/// 标签管理命令
/// </summary>
public sealed class TagCommand
{
    private readonly ISessionTagRepository _tagRepository;
    private readonly SmartTagService _tagService;

    public TagCommand(
        ISessionTagRepository tagRepository,
        SmartTagService tagService)
    {
        _tagRepository = tagRepository;
        _tagService = tagService;
    }

    /// <summary>
    /// 添加标签: /tag add <session-id> <tag> [emoji] [color]
    /// </summary>
    public async Task ExecuteAddAsync(
        Guid sessionId,
        string tagName,
        string? emoji = null,
        string? color = null,
        CancellationToken ct = default)
    {
        try
        {
            var tag = SessionTag.Create(sessionId, tagName, TagSource.User, color, emoji);
            await _tagRepository.AddAsync(tag, ct);

            AnsiConsole.MarkupLine(
                $"[green]✓ 标签已添加:[/] {FormatTag(tag)}"
            );
        }
        catch (Exception ex)
        {
            AnsiConsole.MarkupLine($"[red]✗ 添加标签失败: {ex.Message.EscapeMarkup()}[/]");
        }
    }

    /// <summary>
    /// 移除标签: /tag remove <session-id> <tag>
    /// </summary>
    public async Task ExecuteRemoveAsync(
        Guid sessionId,
        string tagName,
        CancellationToken ct = default)
    {
        try
        {
            await _tagRepository.RemoveAsync(sessionId, tagName, ct);
            AnsiConsole.MarkupLine($"[green]✓ 标签已移除:[/] {tagName}");
        }
        catch (Exception ex)
        {
            AnsiConsole.MarkupLine($"[red]✗ 移除标签失败: {ex.Message.EscapeMarkup()}[/]");
        }
    }

    /// <summary>
    /// 列出标签: /tag list [session-id]
    /// </summary>
    public async Task ExecuteListAsync(
        Guid? sessionId = null,
        CancellationToken ct = default)
    {
        try
        {
            if (sessionId.HasValue)
            {
                await ListSessionTagsAsync(sessionId.Value, ct);
            }
            else
            {
                await ListAllTagsAsync(ct);
            }
        }
        catch (Exception ex)
        {
            AnsiConsole.MarkupLine($"[red]✗ 列出标签失败: {ex.Message.EscapeMarkup()}[/]");
        }
    }

    /// <summary>
    /// 建议标签: /tag suggest <session-id>
    /// </summary>
    public async Task ExecuteSuggestAsync(
        Guid sessionId,
        string sessionTitle,
        CancellationToken ct = default)
    {
        try
        {
            AnsiConsole.MarkupLine("[cyan]💡 生成标签建议中...[/]");

            var suggestions = await _tagService.SuggestFromTitleAsync(sessionTitle, ct);

            if (suggestions.Count == 0)
            {
                AnsiConsole.MarkupLine("[yellow]⚠ 未生成标签建议[/]");
                return;
            }

            // 显示建议
            var table = new Table()
                .Border(TableBorder.Rounded)
                .AddColumn("标签")
                .AddColumn("Emoji")
                .AddColumn("颜色");

            foreach (var suggestion in suggestions)
            {
                table.AddRow(
                    suggestion.Tag,
                    suggestion.Emoji ?? "-",
                    suggestion.Color ?? "-"
                );
            }

            AnsiConsole.Write(table);

            // 询问是否应用
            if (AnsiConsole.Confirm("是否应用这些标签?"))
            {
                await _tagService.ApplySuggestionsAsync(sessionId, suggestions, ct);
                AnsiConsole.MarkupLine("[green]✓ 标签已应用[/]");
            }
        }
        catch (Exception ex)
        {
            AnsiConsole.MarkupLine($"[red]✗ 标签建议失败: {ex.Message.EscapeMarkup()}[/]");
        }
    }

    private async Task ListSessionTagsAsync(Guid sessionId, CancellationToken ct)
    {
        var tags = await _tagRepository.GetBySessionAsync(sessionId, ct);

        if (tags.Count == 0)
        {
            AnsiConsole.MarkupLine("[yellow]⚠ 该会话没有标签[/]");
            return;
        }

        AnsiConsole.MarkupLine($"[cyan]标签列表 ({tags.Count}):[/]");
        foreach (var tag in tags)
        {
            AnsiConsole.MarkupLine($"  {FormatTag(tag)} [{tag.Source}]");
        }
    }

    private async Task ListAllTagsAsync(CancellationToken ct)
    {
        var statistics = await _tagRepository.GetTagStatisticsAsync(ct);

        if (statistics.Count == 0)
        {
            AnsiConsole.MarkupLine("[yellow]⚠ 没有任何标签[/]");
            return;
        }

        var table = new Table()
            .Border(TableBorder.Rounded)
            .AddColumn("标签")
            .AddColumn("使用次数");

        foreach (var (tag, count) in statistics.OrderByDescending(x => x.Value))
        {
            table.AddRow(tag, count.ToString());
        }

        AnsiConsole.Write(table);
        AnsiConsole.MarkupLine($"\n[cyan]总计 {statistics.Count} 个标签[/]");
    }

    private string FormatTag(SessionTag tag)
    {
        var parts = new List<string>();

        if (!string.IsNullOrEmpty(tag.Emoji))
            parts.Add(tag.Emoji);

        parts.Add(tag.Tag);

        if (!string.IsNullOrEmpty(tag.Color))
            parts.Add($"[{tag.Color}]●[/]");

        return string.Join(" ", parts);
    }
}

private async Task HandleTagCommandAsync(string[] parts, CancellationToken ct)
{
    var subcommand = parts[1].ToLower();
    switch (subcommand)
    {
        case "add":
            if (parts.Length < 4)
            {
                AnsiConsole.MarkupLine("[red]✗ 用法: /tag add <session-id> <tag> [emoji] [color][/]");
                return;
            }
            var sessionId = Guid.Parse(parts[2]);
            var tag = parts[3];
            var emoji = parts.Length > 4 ? parts[4] : null;
            var color = parts.Length > 5 ? parts[5] : null;
            await _tagCommand.ExecuteAddAsync(sessionId, tag, emoji, color, ct);
            break;

        case "remove":
            if (parts.Length < 4)
            {
                AnsiConsole.MarkupLine("[red]✗ 用法: /tag remove <session-id> <tag>[/]");
                return;
            }
            await _tagCommand.ExecuteRemoveAsync(Guid.Parse(parts[2]), parts[3], ct);
            break;

        case "list":
            var listSessionId = parts.Length >= 3 ? Guid.Parse(parts[2]) : (Guid?)null;
            await _tagCommand.ExecuteListAsync(listSessionId, ct);
            break;

        case "suggest":
            if (parts.Length < 3)
            {
                AnsiConsole.MarkupLine("[red]✗ 用法: /tag suggest <session-id>[/]");
                return;
            }
            // 需要获取会话标题
            var suggestSessionId = Guid.Parse(parts[2]);
            var session = await _sessionRepository.GetByIdAsync(suggestSessionId, ct);
            if (session != null)
            {
                await _tagCommand.ExecuteSuggestAsync(suggestSessionId, session.Title, ct);
            }
            break;

        default:
            AnsiConsole.MarkupLine($"[red]✗ 未知子命令: {subcommand}[/]");
            ShowTagHelp();
            break;
    }
}

## 技术实现

### 数据模型

```csharp
// SessionTag（复合主键）
SessionTag {
    Guid SessionId,
    string Tag,           // 小写存储
    string? Color,        // RGB 格式
    string? Emoji,
    DateTime CreatedAt,
    TagSource Source      // User | Auto
}

// SearchQuery
SearchQuery {
    string NaturalQuery,
    SearchCriteria Criteria,
    SearchType Type
}

// SearchCriteria
SearchCriteria {
    List<string> Keywords,
    string? RegexPattern,
    List<string> ExactPhrases,
    MessageRole? Role,
    Guid? SessionId,
    DateTimeOffset? StartDate,
    DateTimeOffset? EndDate
}
```

### 数据库架构

```sql
-- SessionTags 表
CREATE TABLE session_tags (
    session_id TEXT NOT NULL,
    tag TEXT NOT NULL,
    color TEXT,
    emoji TEXT,
    created_at TIMESTAMP NOT NULL,
    source TEXT NOT NULL,
    PRIMARY KEY (session_id, tag),
    FOREIGN KEY (session_id) REFERENCES sessions(id) ON DELETE CASCADE
);

CREATE INDEX idx_session_tags_tag ON session_tags(tag);

-- FTS5 虚拟表
CREATE VIRTUAL TABLE messages_fts USING fts5(
    message_id UNINDEXED,
    session_id UNINDEXED,
    content,
    role UNINDEXED,
    created_at UNINDEXED
);

-- 自动同步触发器
CREATE TRIGGER messages_ai AFTER INSERT ON messages BEGIN
    INSERT INTO messages_fts(...) VALUES (new.id, ...);
END;
```

### 服务架构

```
NaturalLanguageQueryService
  ├─ ILLMClient (查询理解)

SmartTagService
  ├─ ILLMClient (标签生成)
  ├─ ISessionTagRepository (持久化)
